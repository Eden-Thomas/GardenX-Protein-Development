// Comparative analysis functions async function runComparativeAnalysis() { const selectedSpecies = []; document.querySelectorAll('.species-checkbox input:checked').forEach(cb => { selectedSpecies.push(cb.value); }); if (selectedSpecies.length < 2) { alert('Please select at least 2 species for comparison'); return; } console.log('Running comparative analysis for:', selectedSpecies); // Show loading state const resultsDiv = document.getElementById('analysis-results'); resultsDiv.style.display = 'block'; resultsDiv.innerHTML = '

Analyzing sequences...

'; try { const response = await fetch('http://localhost:8000/api/analysis/comparative', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ species: selectedSpecies, reference_sequence: document.querySelector('.sequence-input').value || null }) }); if (!response.ok) { throw new Error('Analysis failed'); } const data = await response.json(); console.log('Analysis results:', data); // Display results displayAnalysisResults(data); } catch (error) { console.error('Error:', error); alert('Analysis failed: ' + error.message); } } function displayAnalysisResults(data) { // Restore the results container HTML const resultsHTML = `

Sequence Conservation Analysis

Average Identity ${data.conservation.average_identity}%
Conserved Positions ${data.conservation.conserved_positions}
Variable Positions ${data.conservation.variable_positions}

Thermostability Correlation Analysis

Statistical analysis comparing residue patterns in thermophilic vs mesophilic species

${data.correlations.map(corr => ` `).join('')}
Feature R² Value P-Value Significance
${corr.feature} ${corr.r_squared.toFixed(4)} ${corr.p_value < 0.001 ? '<0.001' : corr.p_value.toFixed(4)} ${getSignificanceLabel(corr.p_value)}

Evidence-Based Mutation Recommendations

Ranked by predicted thermostability impact, conservation score, and statistical significance

${data.recommendations.map(rec => createRecommendationCard(rec)).join('')}

Multiple Sequence Alignment

${renderAlignment(data.alignment)}
`; document.getElementById('analysis-results').innerHTML = resultsHTML; } function getSignificanceClass(pValue) { if (pValue < 0.01) return 'significance-high'; if (pValue < 0.05) return 'significance-medium'; return 'significance-low'; } function getSignificanceLabel(pValue) { if (pValue < 0.001) return 'Highly Significant ***'; if (pValue < 0.01) return 'Significant **'; if (pValue < 0.05) return 'Significant *'; return 'Not Significant'; } function createRecommendationCard(rec) { const confidenceClass = rec.confidence >= 0.8 ? 'high' : rec.confidence >= 0.6 ? 'medium' : 'low'; return `
${rec.from}${rec.position}${rec.to} ${(rec.confidence * 100).toFixed(0)}% Confidence
Rationale: ${rec.rationale}
Predicted ΔΔG: ${rec.delta_stability.toFixed(2)} kcal/mol
Conservation: ${rec.conservation_score.toFixed(2)}
Found in thermophiles: ${rec.thermophile_prevalence}%
Distance from active site: ${rec.distance_from_active}Å
`; } function renderAlignment(alignmentData) { if (!alignmentData || !alignmentData.sequences) { return '

No alignment data available

'; } return alignmentData.sequences.map(seq => `
${seq.species} ${seq.sequence}
`).join(''); }
